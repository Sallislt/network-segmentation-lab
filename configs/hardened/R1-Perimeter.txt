! ============================================================
! R1-Perimeter - edge router (hardened version)
! Changes vs. original:
!  - routed /30 transit link to SW-CORE (the original trunk had no IP, so no VLAN could reach the WAN)
!  - static route back to the internal networks
!  - NAT overload only for the networks allowed to reach the Internet
!  - management hardening: SSH v2 only, local users, VTY restricted to the management VLAN
! Replace every <CHANGE_ME> before use. Not yet re-tested in Packet Tracer.
! ============================================================
enable
configure terminal
hostname R1-Perimeter

! --- Management hardening ---------------------------------------------
service password-encryption
enable secret <CHANGE_ME>
username netadmin privilege 15 secret <CHANGE_ME>
no ip domain-lookup
ip domain-name lab.local
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
ip ssh time-out 60
ip ssh authentication-retries 3
login block-for 120 attempts 3 within 60
banner motd ^Authorized access only. Activity is logged.^

ip access-list standard ACL_MGMT
 permit 192.168.99.0 0.0.0.255
 deny any log

line con 0
 login local
 exec-timeout 10 0
line vty 0 4
 login local
 transport input ssh
 access-class ACL_MGMT in
 exec-timeout 10 0
line vty 5 15
 transport input none

! --- Interfaces ---------------------------------------------------------
interface GigabitEthernet0/0
 description WAN / ISP
 ip address dhcp
 ip nat outside
 no cdp enable
 no shutdown

interface GigabitEthernet0/1
 description Transit to SW-CORE (routed /30)
 ip address 10.0.0.1 255.255.255.252
 ip nat inside
 no shutdown

! Return route to all internal VLANs via the core switch
ip route 192.168.0.0 255.255.0.0 10.0.0.2

! --- NAT: only these networks may reach the Internet ------------------
! Exam PCs (VLAN 20), cameras (VLAN 50), VoIP (VLAN 40) and management (VLAN 99)
! are deliberately not translated: they have no Internet access.
ip access-list standard ACL_NAT
 permit 192.168.10.0 0.0.0.255
 permit 192.168.30.0 0.0.0.255
ip nat inside source list ACL_NAT interface GigabitEthernet0/0 overload

end
write memory
